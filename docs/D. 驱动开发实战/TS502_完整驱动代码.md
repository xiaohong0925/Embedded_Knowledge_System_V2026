## <span class="blue"> 使用说明

- 代码按 Linux 6.x API 编写（i2c probe/remove 新签名、`class_create` 单参数、`DEFINE_SHOW_ATTRIBUTE`、`fsleep` 等）；移植到具体内核版本时个别胶水可能需要微调
- 九篇里的演进关系在总装时做了两处归并，注释里标明：① D.3 的 `fifo_ready` 影子标志在引入 kfifo 后由 `kfifo_is_empty()` 取代——kfifo 长度本身就是合格的内存标志；② D.2 的一次性温度帧 read 演进为 FIFO/kfifo 数据通路，语义以本篇为准
- 设备树完整节点见 D.7；寄存器手册见 D.1

## <span class="blue"> ts502.c 完整代码

```c
// SPDX-License-Identifier: GPL-2.0
/*
 * ts502.c —— TS502 复合传感器驱动（D 扩展 Part 1 贯穿案例完整版）
 *
 * 功能全景：cdev(read/ioctl/poll/fasync) + 中断/轮询双触发 +
 *           kfifo 批量缓冲 + suspend/resume 唤醒源 + sysfs/debugfs
 */

#include <linux/module.h>
#include <linux/i2c.h>
#include <linux/delay.h>
#include <linux/cdev.h>
#include <linux/fs.h>
#include <linux/poll.h>
#include <linux/mutex.h>
#include <linux/spinlock.h>
#include <linux/wait.h>
#include <linux/kfifo.h>
#include <linux/hrtimer.h>
#include <linux/workqueue.h>
#include <linux/interrupt.h>
#include <linux/of.h>
#include <linux/pm.h>
#include <linux/pm_wakeup.h>
#include <linux/debugfs.h>
#include <linux/seq_file.h>
#include <linux/uaccess.h>
#include <linux/idr.h>
#include <linux/slab.h>

/* ==================== 寄存器定义（D.1 手册） ==================== */
#define TS502_REG_CHIP_ID    0x00   /* RO  = 0x50 */
#define TS502_REG_TEMP_H     0x01   /* RO  温度高字节 */
#define TS502_REG_TEMP_L     0x02   /* RO  温度低字节，0.0625°C/LSB */
#define TS502_REG_FIFO_DATA  0x03   /* RO  读一次弹出一个采样点(2B) */
#define TS502_REG_FIFO_STAT  0x04   /* RO  bit[5:0]深度 bit7溢出 */
#define TS502_REG_CTRL       0x05   /* RW  bit0使能 bit1FIFO使能 bit[3:2]采样率 */
#define TS502_REG_INT_STAT   0x06   /* RW1C bit0数据就绪 bit1FIFO半满 */
#define TS502_REG_INT_MASK   0x07   /* RW  中断输出使能 */
#define TS502_REG_PWM_ALARM  0x08   /* RW  报警阈值(°C) */

#define TS502_CHIP_ID        0x50
#define TS502_INT_DATA_READY 0x01
#define TS502_INT_FIFO_HALF  0x02
#define TS502_FIFO_OVFL      0x80

#define TS502_MAX_MINORS     8
#define TS502_FIFO_BUFLEN    4096   /* kfifo：2048 个采样点 */

/* ==================== ioctl ABI（D.2：size/version 演进） ==================== */
struct ts502_cfg {
    __u32 size;         /* 入参：sizeof(struct ts502_cfg)，分版本依据 */
    __u32 version;      /* 出参：驱动支持的 ABI 版本，当前 1 */
    __u32 sample_rate;  /* 0:1Hz 1:10Hz 2:100Hz */
    __s32 alarm;        /* 报警阈值 °C */
};

#define TS502_IOC_MAGIC   't'
#define TS502_IOC_SET_CFG _IOW(TS502_IOC_MAGIC, 0x01, struct ts502_cfg)
#define TS502_IOC_GET_CFG _IOR(TS502_IOC_MAGIC, 0x02, struct ts502_cfg)
#define TS502_IOC_KICK    _IO(TS502_IOC_MAGIC,  0x03)  /* 调试：模拟数据就绪 */

/* ==================== 实例私有数据（逐篇生长） ==================== */
struct ts502_data {
    struct i2c_client *client;

    /* D.2：cdev 接口 */
    struct cdev cdev;
    int minor;
    struct mutex lock;              /* 保护寄存器配置序列 */

    /* D.3：等待与通知 */
    wait_queue_head_t wq;
    struct fasync_struct *fasync;

    /* D.5：轮询后备（INT 缺线的板子） */
    struct hrtimer poll_timer;
    struct work_struct poll_work;
    ktime_t period;
    bool int_connected;

    /* D.6：批量缓冲 */
    struct kfifo fifo;
    struct work_struct fifo_work;   /* D.4 FIFO 半满中断的下游 */

    /* D.7：DT 配置 */
    int alarm;
    u32 sample_rate;

    /* D.8：电源管理上下文 */
    int saved_ctrl;
    int saved_int_mask;
    int saved_alarm;

    /* D.9：统计与调试 */
    unsigned int irq_count;
    unsigned int overflow_count;
    unsigned int wake_count;
    struct dentry *debugfs_dir;
};

/* ==================== 驱动级资源（D.2） ==================== */
static DEFINE_IDA(ts502_ida);
static dev_t ts502_devt;
static struct class *ts502_class;

/* ==================== 通知三动作（D.3/D.4：先置条件，后唤醒） ==================== */
/*
 * 注意：D.3 的 fifo_ready 影子标志已被 kfifo_is_empty() 取代——
 * kfifo 长度本身就是内存里的等待条件，不违反"条件里禁读硬件"。
 */
static void ts502_notify_data(struct ts502_data *data)
{
    wake_up_interruptible(&data->wq);               /* ② 唤醒阻塞/epoll 读者 */
    kill_fasync(&data->fasync, SIGIO, POLL_IN);     /* ③ fasync 通道 */
}

/* ==================== fops：open/release（D.2 per-fd） ==================== */
static int ts502_open(struct inode *inode, struct file *filp)
{
    struct ts502_data *data =
        container_of(inode->i_cdev, struct ts502_data, cdev);

    filp->private_data = data;
    return nonseekable_open(inode, filp);
}

static int ts502_release(struct inode *inode, struct file *filp)
{
    ts502_fasync(-1, filp, 0);          /* 注销异步通知，防野信号 */
    return 0;
}

/* ==================== fops：read（D.3 等待 + D.6 kfifo） ==================== */
static ssize_t ts502_read(struct file *filp, char __user *buf,
                          size_t count, loff_t *ppos)
{
    struct ts502_data *data = filp->private_data;
    unsigned int copied;
    int ret;

    /* 阻塞/非阻塞分支（D.3）：条件 = kfifo 非空 */
    if (kfifo_is_empty(&data->fifo)) {
        if (filp->f_flags & O_NONBLOCK)
            return -EAGAIN;
        ret = wait_event_interruptible(data->wq,
                                       !kfifo_is_empty(&data->fifo));
        if (ret)
            return ret;                 /* -ERESTARTSYS 原样上传 */
    }

    /* kfifo_to_user 直达用户态（D.6）；偶数字节对齐采样帧 */
    ret = kfifo_to_user(&data->fifo, buf, count & ~1UL, &copied);
    if (ret)
        return ret;

    /* EPOLLET 契约（D.3）：业务应 drain 到 EAGAIN；空队列即无新边沿 */
    return copied;
}

/* ==================== fops：poll（D.3） ==================== */
static __poll_t ts502_poll(struct file *filp, poll_table *wait)
{
    struct ts502_data *data = filp->private_data;
    __poll_t mask = 0;

    poll_wait(filp, &data->wq, wait);           /* ① 登记 */
    if (!kfifo_is_empty(&data->fifo))
        mask |= EPOLLIN | EPOLLRDNORM;          /* ② 报告当前状态 */
    return mask;
}

/* ==================== fops：ioctl（D.2 ABI 设计） ==================== */
static long ts502_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    struct ts502_data *data = filp->private_data;
    struct ts502_cfg cfg;
    int ret = 0;

    switch (cmd) {
    case TS502_IOC_SET_CFG:
        if (copy_from_user(&cfg, (void __user *)arg, sizeof(cfg)))
            return -EFAULT;
        if (cfg.size != sizeof(cfg))            /* 版本不匹配（演进时在此分新旧） */
            return -EINVAL;
        if (cfg.sample_rate > 2 || cfg.alarm < 0 || cfg.alarm > 125)
            return -EINVAL;

        mutex_lock(&data->lock);
        /* CTRL：bit0 使能 | bit1 FIFO 使能 | bit[3:2] 采样率 */
        ret = i2c_smbus_write_byte_data(data->client, TS502_REG_CTRL,
                                        (cfg.sample_rate << 2) | 0x03);
        if (ret >= 0)
            ret = i2c_smbus_write_byte_data(data->client,
                                            TS502_REG_PWM_ALARM, cfg.alarm);
        if (ret >= 0) {
            data->sample_rate = cfg.sample_rate;
            data->alarm = cfg.alarm;
            /* 轮询后备模式下同步采样周期（D.5） */
            data->period = ktime_set(0, NSEC_PER_SEC /
                                       (cfg.sample_rate == 2 ? 100 :
                                        cfg.sample_rate == 1 ? 10 : 1));
        }
        mutex_unlock(&data->lock);
        return ret < 0 ? ret : 0;

    case TS502_IOC_GET_CFG:
        memset(&cfg, 0, sizeof(cfg));
        cfg.size = sizeof(cfg);
        cfg.version = 1;
        cfg.sample_rate = data->sample_rate;
        cfg.alarm = data->alarm;
        if (copy_to_user((void __user *)arg, &cfg, sizeof(cfg)))
            return -EFAULT;
        return 0;

    case TS502_IOC_KICK:      /* 调试"踢"（D.3 遗留，保留用于产测）：总装版条件
                                 是 kfifo 非空，KICK 只验证唤醒/poll/fasync 链路，
                                 不注入数据——读者复查条件不满足会重新入睡 */
        ts502_notify_data(data);
        return 0;

    default:
        return -ENOTTY;       /* 不认识的命令：标准返回值 */
    }
}

/* ==================== fops：fasync（D.3） ==================== */
static int ts502_fasync(int fd, struct file *filp, int on)
{
    struct ts502_data *data = filp->private_data;

    return fasync_helper(fd, filp, on, &data->fasync);
}

static const struct file_operations ts502_fops = {
    .owner          = THIS_MODULE,
    .open           = ts502_open,
    .release        = ts502_release,
    .read           = ts502_read,
    .poll           = ts502_poll,
    .unlocked_ioctl = ts502_ioctl,
    .compat_ioctl   = ts502_ioctl,   /* 全定长结构体，直接复用（D.2） */
    .fasync         = ts502_fasync,
    .llseek         = no_llseek,     /* 流式设备显式禁 seek（D.2） */
};

/* ==================== 中断路径（D.4） ==================== */
static irqreturn_t ts502_irq_thread(int irq, void *dev_id)
{
    struct ts502_data *data = dev_id;
    int status;

    /* ① 读中断状态（线程上下文，I2C 合法） */
    status = i2c_smbus_read_byte_data(data->client, TS502_REG_INT_STAT);
    if (status < 0)
        return IRQ_HANDLED;
    if (!(status & (TS502_INT_DATA_READY | TS502_INT_FIFO_HALF)))
        return IRQ_NONE;                        /* 假中断：不是我们的 */

    /* ② 清中断源（RW1C）在分发之前——丢事件第一根因防线 */
    i2c_smbus_write_byte_data(data->client, TS502_REG_INT_STAT, status);

    data->irq_count++;
    if (device_may_wakeup(&data->client->dev)) {
        pm_wakeup_event(&data->client->dev, 0); /* 唤醒事件登记（D.8） */
        data->wake_count++;
    }

    /* ③ 按位分发 */
    if (status & TS502_INT_DATA_READY)
        ts502_notify_data(data);
    if (status & TS502_INT_FIFO_HALF)
        schedule_work(&data->fifo_work);        /* 批量搬运转 workqueue */

    return IRQ_HANDLED;
}

/* ==================== 批量搬运（D.6：FIFO 半满的下游 work） ==================== */
static void ts502_fifo_work(struct work_struct *work)
{
    struct ts502_data *data = container_of(work, struct ts502_data, fifo_work);
    u8 burst[32];                               /* 半满 = 16 帧 × 2B */
    int stat, ret;
    unsigned int len;

    stat = i2c_smbus_read_byte_data(data->client, TS502_REG_FIFO_STAT);
    if (stat < 0)
        return;
    if (stat & TS502_FIFO_OVFL) {
        data->overflow_count++;                 /* D.9 统计埋点 */
        dev_dbg_ratelimited(&data->client->dev, "chip FIFO overflow\n");
    }
    if (!(stat & 0x3f))
        return;

    ret = i2c_smbus_read_i2c_block_data(data->client, TS502_REG_FIFO_DATA,
                                        sizeof(burst), burst);
    if (ret < 0)
        return;

    /* 溢出策略（D.6 显式决策）：满了丢新保旧——kfifo_in 写满即静默截断 */
    len = kfifo_in(&data->fifo, burst, ret);
    if (len < ret)
        data->overflow_count++;
    /* 如需反过来"丢旧保新"：先 kfifo_skip 腾出 ret 字节再 kfifo_in */

    ts502_notify_data(data);
}

/* ==================== 轮询后备（D.5：INT 缺线时 hrtimer 接管） ==================== */
static void ts502_poll_work(struct work_struct *work)
{
    struct ts502_data *data = container_of(work, struct ts502_data, poll_work);
    int stat;

    stat = i2c_smbus_read_byte_data(data->client, TS502_REG_FIFO_STAT);
    if (stat >= 0 && (stat & 0x3f) > 0)
        ts502_fifo_work(&data->fifo_work);      /* 与中断路径数据通路合一 */
}

static enum hrtimer_restart ts502_poll_fn(struct hrtimer *t)
{
    struct ts502_data *data = container_of(t, struct ts502_data, poll_timer);

    schedule_work(&data->poll_work);            /* 原子上下文只转活（D.5） */
    hrtimer_forward_now(t, data->period);       /* 无漂移周期 */
    return HRTIMER_RESTART;
}

/* ==================== sysfs 属性（D.9：一值一属性） ==================== */
static ssize_t alarm_threshold_show(struct device *dev,
                                    struct device_attribute *attr, char *buf)
{
    struct ts502_data *data = dev_get_drvdata(dev);
    int val = i2c_smbus_read_byte_data(data->client, TS502_REG_PWM_ALARM);

    if (val < 0)
        return val;
    return sysfs_emit(buf, "%d\n", val);
}

static ssize_t alarm_threshold_store(struct device *dev,
                                     struct device_attribute *attr,
                                     const char *buf, size_t count)
{
    struct ts502_data *data = dev_get_drvdata(dev);
    int val, ret;

    ret = kstrtoint(buf, 0, &val);
    if (ret || val < 0 || val > 125)
        return -EINVAL;

    mutex_lock(&data->lock);
    ret = i2c_smbus_write_byte_data(data->client, TS502_REG_PWM_ALARM, val);
    if (ret >= 0)
        data->alarm = val;      /* 与 ioctl GET_CFG 的唯一真相同步 */
    mutex_unlock(&data->lock);
    return ret < 0 ? ret : count;
}
static DEVICE_ATTR_RW(alarm_threshold);

static struct attribute *ts502_attrs[] = {
    &dev_attr_alarm_threshold.attr,
    NULL,
};
ATTRIBUTE_GROUPS(ts502);

/* ==================== debugfs（D.9：非 ABI 自由区） ==================== */
static int ts502_regs_show(struct seq_file *s, void *unused)
{
    struct ts502_data *data = s->private;
    static const struct { u8 reg; const char *name; } regs[] = {
        { TS502_REG_CHIP_ID,   "CHIP_ID"   },
        { TS502_REG_TEMP_H,    "TEMP_H"    },
        { TS502_REG_TEMP_L,    "TEMP_L"    },
        /* 0x03 FIFO_DATA 刻意跳过：读它会弹出数据 */
        { TS502_REG_FIFO_STAT, "FIFO_STAT" },
        { TS502_REG_CTRL,      "CTRL"      },
        { TS502_REG_INT_STAT,  "INT_STAT"  },
        { TS502_REG_INT_MASK,  "INT_MASK"  },
        { TS502_REG_PWM_ALARM, "PWM_ALARM" },
    };
    int i, val;

    for (i = 0; i < ARRAY_SIZE(regs); i++) {
        val = i2c_smbus_read_byte_data(data->client, regs[i].reg);
        if (val < 0)
            seq_printf(s, "0x%02x %-9s = <err>\n", regs[i].reg, regs[i].name);
        else
            seq_printf(s, "0x%02x %-9s = 0x%02x\n", regs[i].reg, regs[i].name, val);
    }
    seq_printf(s, "kfifo_len      = %u\n", kfifo_len(&data->fifo));
    seq_printf(s, "irq_count      = %u\n", data->irq_count);
    seq_printf(s, "overflow_count = %u\n", data->overflow_count);
    seq_printf(s, "wake_count     = %u\n", data->wake_count);
    return 0;
}
DEFINE_SHOW_ATTRIBUTE(ts502_regs);

/* ==================== 电源管理（D.8） ==================== */
static int ts502_suspend(struct device *dev)
{
    struct ts502_data *data = dev_get_drvdata(dev);

    mutex_lock(&data->lock);
    /* ① 先保存会丢的寄存器，② 再停采样 */
    data->saved_ctrl = i2c_smbus_read_byte_data(data->client, TS502_REG_CTRL);
    data->saved_int_mask = i2c_smbus_read_byte_data(data->client, TS502_REG_INT_MASK);
    data->saved_alarm = i2c_smbus_read_byte_data(data->client, TS502_REG_PWM_ALARM);
    i2c_smbus_write_byte_data(data->client, TS502_REG_CTRL,
                              data->saved_ctrl & ~0x01);
    /* 轮询后备模式：停 hrtimer，防休眠期间空转 */
    if (!data->int_connected)
        hrtimer_cancel(&data->poll_timer);
    if (device_may_wakeup(dev) && data->int_connected)
        enable_irq_wake(data->client->irq);     /* 唤醒三步之③ */
    mutex_unlock(&data->lock);
    return 0;
}

static int ts502_resume(struct device *dev)
{
    struct ts502_data *data = dev_get_drvdata(dev);

    mutex_lock(&data->lock);
    if (device_may_wakeup(dev) && data->int_connected)
        disable_irq_wake(data->client->irq);    /* 与 suspend 严格对称 */
    /* ① 先恢复静态配置，② 最后使能采样 */
    i2c_smbus_write_byte_data(data->client, TS502_REG_INT_MASK, data->saved_int_mask);
    i2c_smbus_write_byte_data(data->client, TS502_REG_PWM_ALARM, data->saved_alarm);
    i2c_smbus_write_byte_data(data->client, TS502_REG_CTRL, data->saved_ctrl);
    i2c_smbus_write_byte_data(data->client, TS502_REG_INT_STAT, 0xff); /* 清积压事件 */
    /* 硬件状态机决策：FIFO 内容丢弃（D.8 清单第三类），清 kfifo 同步 */
    kfifo_reset(&data->fifo);
    if (!data->int_connected)
        hrtimer_start(&data->poll_timer, data->period, HRTIMER_MODE_REL);
    mutex_unlock(&data->lock);
    return 0;
}
static DEFINE_SIMPLE_DEV_PM_OPS(ts502_pm_ops, ts502_suspend, ts502_resume);

/* ==================== probe（D.1 骨架 + D.2-D.9 逐层装配） ==================== */
static int ts502_probe(struct i2c_client *client)
{
    struct device *dev = &client->dev;
    struct device_node *np = dev->of_node;
    struct ts502_data *data;
    u32 val;
    int id, ret;

    msleep(5);                          /* 手册上电时序（D.1） */

    /* 验明正身（D.1） */
    id = i2c_smbus_read_byte_data(client, TS502_REG_CHIP_ID);
    if (id < 0)
        return dev_err_probe(dev, id, "read CHIP_ID failed\n");
    if (id != TS502_CHIP_ID)
        return dev_err_probe(dev, -ENODEV, "unexpected CHIP_ID 0x%02x\n", id);

    /* devm 骨架（D.1）：内存托管，remove 不碰 */
    data = devm_kzalloc(dev, sizeof(*data), GFP_KERNEL);
    if (!data)
        return -ENOMEM;
    data->client = client;
    i2c_set_clientdata(client, data);

    mutex_init(&data->lock);
    init_waitqueue_head(&data->wq);
    ret = kfifo_alloc(&data->fifo, TS502_FIFO_BUFLEN, GFP_KERNEL);
    if (ret)
        return ret;                     /* 返回值必查（D.1 纪律） */
    INIT_WORK(&data->fifo_work, ts502_fifo_work);
    INIT_WORK(&data->poll_work, ts502_poll_work);
    hrtimer_init(&data->poll_timer, CLOCK_MONOTONIC, HRTIMER_MODE_REL);
    data->poll_timer.function = ts502_poll_fn;

    /* DT 属性读取（D.7）：可选属性配默认值 */
    data->alarm = 90;
    data->sample_rate = 1;
    if (!of_property_read_u32(np, "virtual,alarm-threshold", &val))
        data->alarm = val;
    if (!of_property_read_u32(np, "virtual,default-sample-rate", &val) && val <= 2)
        data->sample_rate = val;
    data->period = ktime_set(0, NSEC_PER_SEC /
                             (data->sample_rate == 2 ? 100 :
                              data->sample_rate == 1 ? 10 : 1));
    /* 三态：no-irq 属性存在或中断号缺失 → 轮询后备（D.5） */
    data->int_connected = !of_property_read_bool(np, "virtual,no-irq") &&
                          client->irq > 0;

    /* 发布 cdev 接口（D.2）：ida → cdev → device_create */
    data->minor = ida_alloc(&ts502_ida, GFP_KERNEL);
    if (data->minor < 0)
        return data->minor;
    cdev_init(&data->cdev, &ts502_fops);
    data->cdev.owner = THIS_MODULE;
    ret = cdev_add(&data->cdev, MKDEV(MAJOR(ts502_devt), data->minor), 1);
    if (ret)
        goto err_ida;
    if (IS_ERR(device_create(ts502_class, dev,
                             MKDEV(MAJOR(ts502_devt), data->minor),
                             NULL, "ts502-%d", data->minor))) {
        ret = -ENODEV;
        goto err_cdev;
    }

    /* 触发源：中断优先，缺线走轮询（D.4/D.5） */
    if (data->int_connected) {
        ret = devm_request_threaded_irq(dev, client->irq, NULL,
                                        ts502_irq_thread, IRQF_ONESHOT,
                                        "ts502", data);
        if (ret)
            goto err_dev;
        i2c_smbus_write_byte_data(client, TS502_REG_INT_MASK,
                                  TS502_INT_DATA_READY | TS502_INT_FIFO_HALF);
    } else {
        hrtimer_start(&data->poll_timer, data->period, HRTIMER_MODE_REL);
    }

    /* 应用 DT 配置：DT 是源头，寄存器是落点（D.7） */
    i2c_smbus_write_byte_data(client, TS502_REG_PWM_ALARM, data->alarm);
    i2c_smbus_write_byte_data(client, TS502_REG_CTRL,
                              (data->sample_rate << 2) | 0x03);

    /* 唤醒能力（D.8 三步之①②，③在 suspend） */
    device_init_wakeup(dev, data->int_connected);

    /* debugfs（D.9） */
    data->debugfs_dir = debugfs_create_dir("ts502", NULL);
    debugfs_create_file("registers", 0444, data->debugfs_dir, data,
                        &ts502_regs_fops);

    dev_info(dev, "TS502 probed (irq=%s)\n",
             data->int_connected ? "connected" : "polled");
    return 0;

err_dev:
    device_destroy(ts502_class, MKDEV(MAJOR(ts502_devt), data->minor));
err_cdev:
    cdev_del(&data->cdev);
err_ida:
    ida_free(&ts502_ida, data->minor);
    return ret;
}

/* ==================== remove（D.1 对称释放 + D.5 销毁纪律） ==================== */
static void ts502_remove(struct i2c_client *client)
{
    struct ts502_data *data = i2c_get_clientdata(client);

    /* 先停触发源、再收下游，全 sync（D.5）： */
    hrtimer_cancel(&data->poll_timer);
    cancel_work_sync(&data->poll_work);
    cancel_work_sync(&data->fifo_work);
    /* devm_request_threaded_irq 的中断在 remove 返回后自动释放 */

    debugfs_remove_recursive(data->debugfs_dir);
    device_init_wakeup(&client->dev, false);
    device_destroy(ts502_class, MKDEV(MAJOR(ts502_devt), data->minor));
    cdev_del(&data->cdev);
    ida_free(&ts502_ida, data->minor);
    kfifo_free(&data->fifo);
    /* devm_kzalloc 的内存由内核释放（D.1） */
}

/* ==================== ID 表与驱动注册（D.1） ==================== */
static const struct of_device_id ts502_of_match[] = {
    { .compatible = "virtual,ts502" },
    { }
};
MODULE_DEVICE_TABLE(of, ts502_of_match);

static const struct i2c_device_id ts502_id[] = {
    { "ts502", 0 },
    { }
};
MODULE_DEVICE_TABLE(i2c, ts502_id);

static struct i2c_driver ts502_driver = {
    .driver = {
        .name = "ts502",
        .of_match_table = ts502_of_match,
        .pm = pm_sleep_ptr(&ts502_pm_ops),
        .dev_groups = ts502_groups,         /* sysfs 属性（D.9） */
    },
    .probe  = ts502_probe,
    .remove = ts502_remove,
    .id_table = ts502_id,
};

/*
 * 手写 init/exit 替代 module_i2c_driver（D.2）：
 * chrdev 区域与 class 是驱动级资源，先于 i2c_add_driver 注册
 */
static int __init ts502_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&ts502_devt, 0, TS502_MAX_MINORS, "ts502");
    if (ret)
        return ret;
    ts502_class = class_create("ts502");
    if (IS_ERR(ts502_class)) {
        unregister_chrdev_region(ts502_devt, TS502_MAX_MINORS);
        return PTR_ERR(ts502_class);
    }
    ret = i2c_add_driver(&ts502_driver);
    if (ret) {
        class_destroy(ts502_class);
        unregister_chrdev_region(ts502_devt, TS502_MAX_MINORS);
    }
    return ret;
}
module_init(ts502_init);

static void __exit ts502_exit(void)
{
    i2c_del_driver(&ts502_driver);
    class_destroy(ts502_class);
    unregister_chrdev_region(ts502_devt, TS502_MAX_MINORS);
}
module_exit(ts502_exit);

MODULE_DESCRIPTION("TS502 composite sensor driver (full version)");
MODULE_AUTHOR("D extension Part 1");
MODULE_LICENSE("GPL");
```
