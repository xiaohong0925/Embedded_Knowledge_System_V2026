## 4. Kernel：Linux内核的启动流程

### 4.1 Kernel镜像格式与解压

当U-Boot执行`bootz`或`booti`命令后，控制权转移到Linux内核。内核启动的第一步是解压（如果内核是压缩的）。

```
┌─────────────────────────────────────────────────────────┐
│                    内核镜像结构（zImage）                   │
├─────────────────────────────────────────────────────────┤
│  0x0000 - 0x0003 │ ARM zImage魔数: 0x016F2818 (小端)    │
│  0x0004 - 0x0023 │ 启动参数（内核加载地址等）             │
│  0x0024 - 0x00EF │ 解压代码（自解压stub）                 │
│  0x00F0 - ...    │ 压缩的内核镜像（gzip/bzip2/lzma/lz4）  │
└─────────────────────────────────────────────────────────┘

自解压流程：
1. U-Boot跳转到zImage的入口点
2. zImage的前0x24字节是解压stub代码
3. 解压stub在原地或指定地址解压内核
4. 解压完成后跳转到真正的内核入口（stext）
```

```asm
/* arch/arm/boot/compressed/head.S — zImage自解压入口 */

_start:
    /* 保存寄存器（U-Boot传递的参数） */
    /* r0 = 0 (必须为0) */
    /* r1 = machine type (ATAGS，现在用DTB) */
    /* r2 = DTB物理地址 */
    mov r7, r0
    mov r8, r1
    mov r9, r2
    
    /* 检测我们运行在哪 */
    bl  __lookup_processor_type    /* 查找CPU类型 */
    bl  __lookup_machine_type      /* 查找机器类型（legacy） */
    
    /* 计算解压目标地址 */
    /* 如果启用了CONFIG_AUTO_ZRELADDR，自动计算 */
    /* 否则使用CONFIG_PHYS_OFFSET */
    
    /* 创建页表（用于解压期间的MMU映射） */
    bl  create_page_tables
    
    /* 使能MMU */
    bl  __enable_mmu
    
    /* 调用解压函数 */
    bl  decompress_kernel        /* 解压内核 */
    
    /* 关闭MMU和缓存 */
    bl  __mmu_off
    
    /* 跳转到解压后的内核 */
    b   __enter_kernel
    
__enter_kernel:
    /* 恢复寄存器 */
    mov r0, #0                   /* 必须为0 */
    mov r1, r8                   /* machine type */
    mov r2, r9                   /* DTB地址 */
    
    /* 跳转 */
    mov pc, r4                   /* r4 = 解压后的内核入口 */
```

### 4.2 start_kernel() — Linux内核的主入口

解压完成后，内核跳转到`stext`（ARM）或`_stext`（ARM64），最终汇聚到`start_kernel()`函数。

```c
/* init/main.c — start_kernel() — Linux内核的C语言入口 */

asmlinkage __visible void __init start_kernel(void)
{
    char *command_line;
    char *after_dashes;
    
    /* === 第一阶段：架构无关的最早期初始化 === */
    
    set_task_stack_end_magic(&init_task);  /* 设置init任务的栈 */
    smp_setup_processor_id();              /* 设置当前CPU ID */
    cgroup_init_early();                   /* 早期cgroup初始化 */
    local_irq_disable();                   /* 关本地中断 */
    early_boot_irqs_disabled = true;
    
    /* === 第二阶段：架构相关初始化 === */
    
    boot_cpu_init();                       /* 标记boot CPU */
    page_address_init();                   /* 页地址初始化 */
    setup_arch(&command_line);             /* ★ 架构特定初始化 ★ */
    /* setup_arch是各架构的核心初始化函数：
     * ARM: arch/arm/kernel/setup.c
     * ARM64: arch/arm64/kernel/setup.c
     * x86: arch/x86/kernel/setup.c
     * 主要工作：
     *   - 解析设备树（DTB）
     *   - 设置内存布局（memory banks）
     *   - 初始化MMU/页表
     *   - 设置定时器
     *   - 解析bootargs
     */
    
    /* === 第三阶段：内存管理初始化 === */
    
    mm_init_cpumask(&init_mm);             /* 初始化init_mm的CPU掩码 */
    setup_per_cpu_areas();                 /* 设置per-CPU变量区域 */
    trap_init();                           /* 异常/中断向量初始化 */
    mm_init();                             /* 内存管理子系统初始化 */
    /* mm_init内部：
     *   - page_alloc_init()   页分配器初始化
     *   - mem_init()          内存最终初始化
     *   - kmem_cache_init()   slab分配器初始化
     *   - vmalloc_init()      vmalloc初始化
     */
    
    /* === 第四阶段：调度器和IRQ初始化 === */
    
    sched_init();                          /* 进程调度器初始化 */
    preempt_disable();                     /* 禁止抢占 */
    radix_tree_init();                     /* 基数树初始化 */
    irq_early_init();                      /* 早期IRQ初始化 */
    tick_init();                           /* tick初始化 */
    init_timers();                         /* 定时器子系统 */
    hrtimers_init();                       /* 高精度定时器 */
    softirq_init();                        /* 软中断 */
    timekeeping_init();                    /* 时间keeping */
    time_init();                           /* 架构特定时间初始化 */
    
    /* === 第五阶段：控制台和printk === */
    
    console_init();                        /* 控制台初始化 */
    /* 从此printk可以输出到控制台 */
    /* 之前的printk存储在log buffer中 */
    
    if (panic_later)
        panic("%s", panic_later);         /* 处理早期panic */
    
    /* === 第六阶段：内核子系统初始化 === */
    
    mem_init();                            /* 内存最终初始化 */
    kmem_cache_init();                     /* slab分配器 */
    calibrate_delay();                     /* BogoMIPS计算 */
    pidhash_init();                        /* PID哈希表 */
    vfs_caches_init_early();               /* VFS缓存早期初始化 */
    pagecache_init();                      /* 页缓存 */
    
    /* === 第七阶段：rest_init — 创建init进程和kthreads === */
    
    rest_init();                           /* ★ 创建PID 1 ★ */
    /* rest_init永不返回 */
}
```

### 4.3 setup_arch() — 架构特定初始化（ARM）

```c
/* arch/arm/kernel/setup.c — setup_arch() */

void __init setup_arch(char **cmdline_p)
{
    const struct machine_desc *mdesc;
    
    /* 1. 处理器初始化 */
    processor.id = read_cpuid_id();       /* 读取CPU ID */
    processor.ctrl = read_cpuid_ctrl();   /* 读取控制寄存器 */
    
    /* 2. 查找机器描述符 */
    /* 从设备树兼容字符串匹配 machine_desc */
    mdesc = setup_machine_fdt(__atags_pointer); /* __atags_pointer = DTB地址 */
    if (!mdesc)
        mdesc = lookup_machine_type(processor.id); /* fallback到ATAGS */
    machine_desc = mdesc;
    
    /* 3. 设置内存布局 */
    /* 从设备树解析/memory节点 */
    setup_machine_tags(mdesc, &tags);
    parse_tags(tags);
    
    /* 4. 解析命令行参数 */
    /* 从设备树/chosen/bootargs读取 */
    *cmdline_p = default_command_line;
    strlcpy(boot_command_line, *cmdline_p, COMMAND_LINE_SIZE);
    parse_early_param();                   /* 解析early参数 */
    
    /* 5. 初始化分页 */
    paging_init(mdesc);                    /* 设置页表 */
    
    /* 6. 请求标准内存区域 */
    request_standard_resources(&meminfo, mdesc);
    
    /* 7. 初始化IRQ */
    init_irq();                            /* IRQ子系统 */
    
    /* 8. 初始化timer */
    time_init();                           /* 架构特定timer */
    
    /* 9. 控制台初始化 */
    console_init();
    
    /* 10. 缓存和TLB */
    cacheid_init();                        /* 缓存类型检测 */
    
#ifdef CONFIG_OF
    /* 11. 设备树展开（unflatten） */
    unflatten_device_tree();               /* 将FDT展开为内核设备树 */
#endif
    
    /* 12. 遍历设备树，注册平台设备 */
    of_platform_populate(NULL, of_default_bus_match_table, NULL, NULL);
}
```

### 4.4 rest_init() — 创建init进程

```c
/* init/main.c — rest_init() */

noinline void __init_refok rest_init(void)
{
    int pid;
    
    /* 1. 创建kernel_init线程 — 这就是PID 1（init进程） */
    /* kernel_init最终会启动用户空间的/sbin/init */
    pid = kernel_thread(kernel_init, NULL, CLONE_FS | CLONE_SIGHAND);
    
    /* 2. 创建kthreadd — 内核线程的守护进程（PID 2） */
    pid = kernel_thread(kthreadd, NULL, CLONE_FS | CLONE_FILES);
    kthreadd_task = find_task_by_pid_ns(pid, &init_pid_ns);
    
    /* 3. 当前进程变成idle进程（PID 0） */
    init_idle_bootup_task(current);
    schedule_preempt_disabled();
    cpu_startup_entry(CPUHP_ONLINE);       /* idle循环 */
}

/* kernel_init — PID 1的执行路径 */
static int __ref kernel_init(void *unused)
{
    /* 1. 等待kthreadd准备就绪 */
    wait_for_completion(&kthreadd_done);
    
    /* 2. 设置init进程的命名空间 */
    kernel_init_freeable();
    
    /* 3. 初始化 SMP（如果启用） */
    smp_init();
    
    /* 4. 调度器策略初始化 */
    sched_init_smp();
    
    /* 5. 工作队列初始化 */
    workqueue_init();
    
    /* 6. 初始化驱动模型 */
    driver_init();
    
    /* 7. ★ 初始化所有内置驱动（initcall）★ */
    /* 这是内核驱动probe的关键阶段！ */
    do_initcalls();
    /* do_initcalls按level顺序执行：
     *   early_initcall   — 最早期（0级）
     *   core_initcall    — 核心子系统（1级）
     *   postcore_initcall — 核心后（2级）
     *   arch_initcall    — 架构特定（3级）
     *   subsys_initcall  — 子系统（4级）
     *   fs_initcall      — 文件系统（5级）
     *   device_initcall  — 设备驱动（6级）
     *   late_initcall    — 晚期（7级）
     */
    
    /* 8. 初始化tmpfs */
    init_tmpfs();
    
    /* 9. 打开/dev/console作为stdin/stdout/stderr */
    /* 如果/dev/console不存在，会报错但继续 */
    if (sys_open((const char __user *) "/dev/console", O_RDWR, 0) < 0)
        printk(KERN_WARNING "Warning: unable to open an initial console.\n");
    
    /* 10. 尝试执行用户空间init程序 */
    /* 按以下顺序尝试： */
    if (execute_command) {
        /* 如果指定了init=参数 */
        ret = run_init_process(execute_command);
    }
    if (!ret) ret = run_init_process("/sbin/init");
    if (!ret) ret = run_init_process("/etc/init");
    if (!ret) ret = run_init_process("/bin/init");
    if (!ret) ret = run_init_process("/bin/sh");
    
    /* 全部失败 → panic */
    panic("No working init found. Try passing init= option to kernel."
          " See Linux Documentation/admin-guide/init.rst");
}
```

## 5. Rootfs：用户空间的起点

### 5.1 initramfs vs 直接挂载rootfs

内核启动的最后一步是挂载rootfs并执行init程序。这里有两种主要方式：

```
方式一：直接挂载rootfs（最常见）
┌─────────┐    ┌─────────┐    ┌─────────────┐    ┌─────────┐
│ Kernel  │───→│ 挂载    │───→│ 执行/sbin/init│───→│ systemd │
│ 启动    │    │ /dev/xx │    │             │    │/sysvinit│
└─────────┘    └─────────┘    └─────────────┘    └─────────┘

方式二：initramfs（两阶段启动）
┌─────────┐    ┌─────────────┐    ┌─────────────┐    ┌─────────┐
│ Kernel  │───→│ 挂载        │───→│ 执行/init   │───→│ switch  │
│ 启动    │    │ initramfs   │    │（准备rootfs）│    │ _root   │
└─────────┘    │ (tmpfs)     │    │             │    │         │
               └─────────────┘    └─────────────┘    └────┬────┘
                                                          │
                              ┌───────────────────────────┘
                              ▼
                         ┌─────────┐    ┌─────────┐
                         │ 挂载    │───→│ 执行    │
                         │ 真实root│    │ 真实init│
                         └─────────┘    └─────────┘
```

**initramfs的用途**：

1. **驱动模块化**：rootfs所在存储介质的驱动未编译进内核，需要initramfs先加载驱动模块
2. **LVM/RAID/加密**：复杂的存储堆栈需要用户空间工具来组装
3. **网络rootfs**：NFS/ iSCSI root需要用户空间配置网络
4. **恢复模式**：提供一个最小的shell环境用于系统恢复
5. **A/B升级**：在切换rootfs前执行升级逻辑

**initramfs的构建**：

```bash
# 使用initramfs-tools（Debian/Ubuntu）
$ update-initramfs -c -k $(uname -r)
# 产物: /boot/initrd.img-$(uname -r)

# 使用dracut（Fedora/RHEL/openSUSE）
$ dracut -f /boot/initramfs-$(uname -r).img $(uname -r)

# 使用Yocto/Buildroot自定义
# 配置CONFIG_INITRAMFS_SOURCE指向cpio归档或目录

# 查看initramfs内容
$ mkdir /tmp/initramfs && cd /tmp/initramfs
$ zcat /boot/initrd.img-$(uname -r) | cpio -idmv
$ ls -la
```

### 5.2 init进程的选择与启动

内核会按优先级尝试多个init程序位置：

```c
/* init/main.c — 用户空间init的执行顺序 */

static int __ref kernel_init(void *unused)
{
    /* ... 前面的初始化 ... */
    
    /* 如果内核参数指定了init= */
    if (execute_command) {
        ret = run_init_process(execute_command);
        if (!ret)
            return 0;
        panic("Requested init %s failed (error %d).", execute_command, ret);
    }
    
    /* 默认搜索路径 */
    if (!run_init_process("/sbin/init"))
        return 0;
    if (!run_init_process("/etc/init"))
        return 0;
    if (!run_init_process("/bin/init"))
        return 0;
    if (!run_init_process("/bin/sh"))
        return 0;
    
    panic("No working init found.");
}
```

**三种主流init系统对比**：

| 特性 | systemd | SysV init | BusyBox init |
|------|---------|-----------|--------------|
| 启动速度 | 中等（并行启动） | 慢（串行） | 快（最简单） |
| 内存占用 | 大（~20-50MB） | 中（~5-10MB） | 小（~1-2MB） |
| 功能丰富度 | 极高（cgroup、log、network、timer…） | 低 | 低 |
| 依赖管理 | 自动（socket/dbus激活） | 手动（启动脚本顺序） | 手动（inittab） |
| 调试难度 | 中（systemd-analyze） | 低（简单脚本） | 低 |
| 嵌入式适用 | 资源充足时可用 | 传统系统 | ★ 资源受限首选 ★ |
| 启动日志 | journalctl | syslog | dmesg/文件 |

### 5.3 BusyBox init的启动流程

资源受限的嵌入式系统通常使用BusyBox init，其配置通过`/etc/inittab`：

```bash
# /etc/inittab — BusyBox init配置

# 系统启动时执行
::sysinit:/etc/init.d/rcS

# 控制台登录
::respawn:/sbin/getty -L ttymxc0 115200 vt100

# Ctrl+Alt+Del重启
::ctrlaltdel:/sbin/reboot

# 关机时执行
::shutdown:/etc/init.d/rcK

# 电源故障
::powerfail:/sbin/poweroff
```

```bash
# /etc/init.d/rcS — 系统启动脚本（典型嵌入式）
#!/bin/sh

# 1. 挂载基本文件系统
mount -t proc proc /proc
mount -t sysfs sysfs /sys
mount -t devtmpfs devtmpfs /dev
mount -t tmpfs tmpfs /tmp
mount -t tmpfs tmpfs /run

# 2. 创建必要的设备节点
mkdir -p /dev/pts
mount -t devpts devpts /dev/pts

# 3. 设置主机名
hostname $(cat /etc/hostname 2>/dev/null || echo "embedded")

# 4. 配置网络
ifconfig lo 127.0.0.1 up
ifconfig eth0 192.168.1.100 netmask 255.255.255.0 up

# 5. 加载内核模块（如果驱动未内置）
modprobe imx_sdma
modprobe galcore  # GPU驱动

# 6. 启动主应用
echo "Starting main application..."
/usr/bin/my_app &

# 7. 可选：启动sshd
dropbear -B &
```

> ⚠️ **陷阱 — initramfs和rootfs混淆（initramfs中的init脚本有问题）**：某系统使用initramfs作为过渡，initramfs中的`/init`脚本负责解密磁盘后switch_root。但由于脚本中的一个条件判断错误（`if [ -f /dev/mapper/rootfs ]`而非`-b`），导致在特定时序下解密未完成就尝试切换，表现为"kernel panic - not syncing: Attempted to kill init"。排查了三天才发现是initramfs的问题而非内核问题。关键教训：(1) initramfs中的`/init`必须正确处理所有错误路径；(2) 使用`set -e`让shell脚本在错误时退出；(3) 在initramfs中加入调试模式（如通过内核参数`rd.debug`启用详细日志）。

## 7. 启动时间分解与优化

### 7.1 各阶段典型耗时数据

| 阶段 | 未优化耗时 | 优化后耗时 | 占比（未优化） | 优化方法 |
|------|-----------|-----------|--------------|---------|
| BootROM | 50-300ms | 50-200ms | 2-5% | 降低存储CLK频率（提高可靠性） |
| SPL（DDR初始化） | 100-500ms | 50-200ms | 3-8% | 优化DDR参数、启用cache |
| U-Boot | 300ms-3s | 0（Falcon Mode） | 15-30% | ★ Falcon Mode（最大收益）★ |
| Kernel解压+初始化 | 500ms-5s | 200ms-1s | 30-50% | 裁剪、quiet、initcall_debug |
| rootfs挂载 | 200ms-2s | 100-500ms | 5-15% | 选用更快的存储、精简rootfs |
| 用户空间（systemd） | 1-10s | 200ms-2s | 30-50% | 服务裁剪、依赖优化 |
| **总计** | **3-20s** | **1-4s** | **100%** | |

### 7.3 瓶颈识别方法

```bash
# 1. 确定哪个阶段是瓶颈
# 如果总启动时间 > 10秒：
#   - Kernel阶段 > 3秒 → 驱动过多，需要裁剪
#   - Userspace阶段 > 5秒 → 服务过多，需要裁剪

$ dmesg > /tmp/dmesg.log
# 找出时间跳跃点（耗时长的操作）

    awk '{match($0, /\[([0-9]+\.[0-9]+)\]/, a); print a[1], $0}' | \
    awk '{delta=$1-prev; prev=$1; if(delta>0.1) print delta, $0}'

$ systemd-analyze blame | head -20

## 本节总结

### 核心要点回顾

```
Kernel→Rootfs关键认知：
1. zImage自解压 → stext → start_kernel() → setup_arch() → rest_init()
2. setup_arch解析DTB、设置内存布局、初始化MMU/IRQ/Timer
3. rest_init创建PID 1(kernel_init)和PID 2(kthreadd)
4. kernel_init执行do_initcalls（驱动probe关键阶段）→ 执行/sbin/init
5. init进程启动systemd或BusyBox init，完成用户空间初始化
6. rootfs挂载通过VFS层完成，rootwait参数确保设备就绪
```

### 关键决策清单

1. **root=参数可靠性**：使用UUID或PARTUUID替代设备号，防止设备号变化
2. **initramfs必要性**：存储驱动模块化或LVM/加密时必须使用initramfs
3. **init系统选择**：资源受限选BusyBox init，功能需求高选systemd
4. **启动时间优化**：initcall_debug定位慢驱动，systemd-analyze定位慢服务
