    16.4.3 U-Boot完整启动流程¶

    版本: V2026-draft 目标读者: 5-8年经验的嵌入式Linux工程师 前置知识: 16.4.1 BootROM + 16.4.2 SPL 核心决策问题: "U-Boot提示Failed to load zImage——是环境变量问题还是存储问题？"

    3. U-Boot：完整的Bootloader¶

    3.1 U-Boot的完整启动流程¶

    当SPL成功初始化DDR并将U-Boot完整版加载到DDR后，控制权转移到U-Boot。U-Boot的启动流程比SPL复杂得多，可以分为两大阶段：

    ┌──────────────────────────────────────────────────────────────────┐
│                    U-Boot启动流程                                 │
├──────────────────────────────────────────────────────────────────┤
│                                                                  │
│  board_init_f() — "f"代表Flash（此时DRAM已可用但 BSS未清零）      │
│  ├── initf_malloc()        — 初始化早期malloc                   │
│  ├── initf_dm()            — 初始化Driver Model                  │
│  ├── initf_console()       — 初始化控制台（串口输出可用）         │
│  ├── fdtdec_setup()        — 解析设备树（如果启用OF_CONTROL）     │
│  ├── initf_dm_devices()    — 预初始化DM设备                      │
│  ├── arch_cpu_init()       — SoC特定CPU初始化                    │
│  ├── mark_bootstage()      — 记录启动阶段标记                    │
│  └── board_early_init_f()  — 板级早期初始化                      │
│                                                                  │
│  [relocate_code] — 将U-Boot从加载地址复制到链接地址              │
│                                                                  │
│  board_init_r() — "r"代表RAM（BSS已清零，完整C环境）             │
│  ├── initr_malloc()        — 初始化完整malloc                    │
│  ├── initr_console()       — 初始化完整控制台                    │
│  ├── initr_dm_devices()    — 初始化所有DM设备                    │
│  ├── initr_env()           — 加载环境变量                        │
│  ├── initr_mmc()           — 初始化MMC/SD控制器                  │
│  ├── initr_net()           — 初始化网络                          │
│  ├── initr_usb()           — 初始化USB                           │
│  ├── initr_jumptable()     — 初始化函数跳转表                    │
│  ├── initr_cmd()           — 注册命令集                          │
│  ├── initr_bootstage()     — 报告启动阶段耗时                    │
│  └── run_main_loop()       — 进入主循环                          │
│       ├── bootdelay        — 等待用户中断（默认3秒）              │
│       └── autoboot_command() — 执行bootcmd                       │
│            ├── load kernel image (bootm/bootz/booti)            │
│            ├── load device tree blob (FDT)                       │
│            ├── setup boot arguments                              │
│            └── boot the kernel                                   │
│                                                                  │
└──────────────────────────────────────────────────────────────────┘

    3.2 board_init_f() 代码分析¶

    /* common/board_f.c — board_init_f() */

void board_init_f(ulong boot_flags)
{
    gd_t *new_gd;

    /* 初始化全局数据指针 gd */
    /* gd（global data）是U-Boot的全局数据结构，贯穿整个启动过程 */
    new_gd = &gdata;  /* 使用SRAM中的静态实例 */
    memset(new_gd, '\0', sizeof(gd_t));
    gd = new_gd;

    /* 设置标志位 */
    gd->flags = boot_flags;
    gd->have_console = 0;

    /* 执行init_sequence_f[]数组中的所有初始化函数 */
    for (init_fnc_ptr = init_sequence_f; *init_fnc_ptr; ++init_fnc_ptr) {
        if ((*init_fnc_ptr)() != 0)
            hang();  /* 初始化失败，死循环 */
    }

    /* 计算重定位地址 */
    /* U-Boot可能需要将自己从加载地址复制到DDR的更高地址 */
    gd->relocaddr = calculate_relocation_addr();

    /* 跳过重定位 */
    relocate_code(gd->start_addr_sp, gd->relocaddr);
}

/* init_sequence_f[] — board_init_f阶段的初始化函数表 */
static const init_fnc_t init_sequence_f[] = {
    setup_mon_len,           /* 设置U-Boot镜像长度 */
    initf_malloc,            /* 早期malloc（使用SRAM） */
    initf_console,           /* 控制台初始化 */
    fdtdec_setup,            /* Flat Device Tree解析 */
    initf_dm,                /* Driver Model初始化 */
    arch_cpu_init,           /* 架构特定CPU初始化 */
    mach_cpu_init,           /* SoC特定初始化 */
    initf_dm_devices,        /* 预初始化DM设备 */
    env_init,                /* 环境变量框架初始化 */
    init_baud_rate,          /* 波特率设置 */
    serial_init,             /* 串口硬件初始化 */
    console_init_f,          /* 控制台（前半部分） */
    fdtdec_prepare_fda,      /* FDT准备 */
    display_options,         /* 打印U-Boot版本 */
    display_text_info,       /* 打印文本信息 */
    print_cpuinfo,           /* 打印CPU信息 */
    show_board_info,         /* 打印板级信息 */
    INIT_FUNC_WATCHDOG_INIT, /* 看门狗初始化 */
    INIT_FUNC_WATCHDOG_RESET,/* 看门狗复位 */
    init_func_i2c,           /* I2C初始化 */
    init_func_spi,           /* SPI初始化 */
    announce_dram_init,      /* 宣布DDR初始化 */
    dram_init,               /* DDR容量检测 */

    3.3 board_init_r() 代码分析¶

    /* common/board_r.c — board_init_r() */

void board_init_r(gd_t *new_gd, ulong dest_addr)
{
    gd = new_gd;

    /* 重新初始化BSS段 */
    memset(__bss_start, '\0', __bss_end - __bss_start);

    /* 重新初始化全局数据 */
    gd->flags |= GD_FLG_RELOC;  /* 标记已重定位 */
    gd->reloc_off = dest_addr - CONFIG_SYS_TEXT_BASE;

    /* 执行init_sequence_r[]数组 */
    for (init_fnc_ptr = init_sequence_r; *init_fnc_ptr; ++init_fnc_ptr) {
        if ((*init_fnc_ptr)() != 0)
            hang();
    }

    /* 不应该到达这里 */
}

    3.4 环境变量加载机制¶

    U-Boot的环境变量是启动配置的核心，存储了bootcmd、bootargs、网络参数等关键配置。

    /* env/common.c — 环境变量加载 */

int env_init(void)
{
    struct env_driver *drv;
    int ret;

    /* 1. 查找环境变量存储驱动 */
    drv = env_driver_lookup_default();

    /* 2. 从存储介质加载 */
    /* 优先级：MMC → NAND → SPI-NOR → EEPROM → 内存 */
    ret = drv->load();

    if (ret == -ENOENT) {
        /* 环境变量不存在，使用编译时默认值 */
        gd->env_addr = (ulong)&default_environment[0];
        gd->env_valid = ENV_VALID;
        return 0;
    }

    if (ret < 0) {
        /* 加载失败（如CRC校验失败） */
        /* 使用默认环境变量，并标记为需要保存 */
        gd->env_addr = (ulong)&default_environment[0];
        gd->env_valid = ENV_INVALID;
        return 0;
    }

    /* 3. CRC校验 */
    if (!env_crc_ok()) {
        printf("*** Warning - bad CRC, using default environment\n");
        gd->env_addr = (ulong)&default_environment[0];
        gd->env_valid = ENV_INVALID;
    }

    return 0;
}

/* 环境变量的存储位置（以eMMC为例） */
/* 通常存储在eMMC的特定偏移或独立分区 */
#define CONFIG_ENV_OFFSET      0xC0000    /* 偏移768KB */
#define CONFIG_ENV_SIZE        0x20000    /* 大小128KB */
#define CONFIG_ENV_IS_IN_MMC   1

    环境变量的关键配置：

    # U-Boot环境变量的典型配置
=> printenv

bootdelay=3                          # 启动延迟3秒
baudrate=115200                      # 串口波特率
ipaddr=192.168.1.100                 # 本机IP
gatewayip=192.168.1.1                # 网关
netmask=255.255.255.0                # 子网掩罩

# ★ bootcmd — 自动启动命令 ★
bootcmd=mmc dev 0; \
       mmc rescan; \
       fatload mmc 0:1 ${loadaddr} zImage; \
       fatload mmc 0:1 ${fdtaddr} ${soc}-${board}.dtb; \
       bootz ${loadaddr} - ${fdtaddr}

# bootcmd逐行解读：
# mmc dev 0      — 选择MMC设备0（eMMC）
# mmc rescan     — 重新扫描MMC设备
# fatload ...    — 从FAT分区加载zImage到${loadaddr}
# fatload ...    — 从FAT分区加载DTB到${fdtaddr}
# bootz ...      — 启动压缩的zImage内核

# ★ bootargs — 传递给内核的命令行参数 ★
bootargs=console=ttymxc0,115200 rw root=/dev/mmcblk0p2 rootwait
# bootargs解读：
# console=ttymxc0,115200  — 控制台设备为ttymxc0，波特率115200
# rw                      — 以读写模式挂载rootfs
# root=/dev/mmcblk0p2     — rootfs在第2个分区
# rootwait                — 等待root设备就绪

    ⚠️ 陷阱 — U-Boot环境变量损坏导致无法找到kernel：某产品在现场偶发启动失败，U-Boot提示"Failed to load 'zImage'"，但实际文件存在。排查发现环境变量区的CRC偶发校验失败（eMMC的bit flip），U-Boot回退到默认环境变量，但默认的bootcmd指向的是错误的分区号（mmc 0:1 vs 实际的mmc 0:2）。解决方案：(1) 在U-Boot配置中增加CONFIG_ENV_REDUND启用冗余环境变量（双份存储+CRC）；(2) 在生产烧录时确保默认环境变量也是正确的；(3) 在kernel启动脚本中加入fallback逻辑。

    3.5 设备树（DTB）加载机制¶

    现代U-Boot使用设备树（Device Tree）来描述硬件，内核启动时需要传递正确的DTB。

    /* cmd/bootz.c — bootz命令的执行流程 */

int do_bootz(struct cmd_tbl *cmdtp, int flag, int argc, char *const argv[])
{
    ulong kernel_addr, initrd_addr, dtb_addr;
    struct bootm_headers images;

    /* 1. 解析参数 */
    kernel_addr = argv[1] ? simple_strtoul(argv[1], NULL, 16) : image_load_addr;
    initrd_addr = argv[2] && strcmp(argv[2], "-") ? simple_strtoul(argv[2], NULL, 16) : 0;
    dtb_addr = argv[3] ? simple_strtoul(argv[3], NULL, 16) : 0;

    /* 2. 验证zImage魔数 */
    if (!valid_zimage_magic(kernel_addr)) {
        printf("Bad Linux ARM zImage magic!\n");
        return 1;
    }

    /* 3. 设置bootm images结构 */
    memset(&images, '\0', sizeof(images));
    images.ep = kernel_addr;           /* 入口点 */
    images.os.start = kernel_addr;     /* OS镜像起始 */
    images.os.type = IH_OS_LINUX;      /* OS类型 = Linux */
    images.os.comp = IH_COMP_GZIP;     /* 压缩方式 = gzip */

    /* 4. 加载/验证DTB */
    if (dtb_addr) {
        /* 用户指定了DTB地址 */
        images.ft_addr = (void *)dtb_addr;
    } else {
        /* 尝试从bootm环境变量或FIT镜像获取DTB */
        images.ft_addr = locate_dtb(&images);
    }

    /* 5. 验证DTB */
    if (fdt_check_header(images.ft_addr) != 0) {
        printf("Invalid device tree blob!\n");
        return 1;
    }

    /* 6. 修改DTB — 插入内核命令行参数和内存信息 */
    fdt_setup(&images);
    /* 这会将bootargs插入到DTB的/chosen节点 */
    /* 将内存布局插入到DTB的/memory节点 */

    /* 7. 关闭缓存、MMU，准备跳转到内核 */
    boot_prep_linux(&images);

    /* 8. 跳转到内核 */
    boot_jump_linux(&images, 0);

    return 0;  /* 不应该到达 */
}

    DTB修改的关键代码：

    /* arch/arm/lib/bootm.c — 在跳转到内核前修改DTB */

int fdt_setup(struct bootm_headers *images)
{
    void *fdt = images->ft_addr;
    int nodeoffset;

    /* 1. 展开FDT（如果压缩） */
    fdt_open_into(fdt, fdt, fdt_totalsize(fdt) + 4096);

    /* 2. 找到/chosen节点 */
    nodeoffset = fdt_path_offset(fdt, "/chosen");
    if (nodeoffset < 0) {
        /* 创建/chosen节点 */
        nodeoffset = fdt_add_subnode(fdt, 0, "chosen");
    }

    /* 3. 插入bootargs（从环境变量读取） */
    char *bootargs = env_get("bootargs");
    if (bootargs) {
        fdt_setprop_string(fdt, nodeoffset, "bootargs", bootargs);
    }

    /* 4. 插入initrd信息（如果使用initramfs） */
    if (images->rd_start) {
        fdt_setprop_u32(fdt, nodeoffset, "linux,initrd-start", images->rd_start);
        fdt_setprop_u32(fdt, nodeoffset, "linux,initrd-end", images->rd_end);
    }

    /* 5. 设置/machine-model */
    fdt_setprop_string(fdt, 0, "model", CONFIG_SYS_BOARD);

    return 0;
}

    3.6 Kernel加载命令详解（bootm/bootz/booti）¶

    U-Boot提供了多个命令来加载不同类型的内核镜像：

    命令 内核格式 ARM 32位 ARM 64位 使用场景
    bootm uImage（旧格式） ✓ ✓ legacy系统
    bootz zImage ✓ — ARM 32位主流
    booti Image（Image.gz） — ✓ ARM 64位主流
    bootelf ELF格式 ✓ ✓ 特殊场景
    bootvx VxWorks ✓ ✓ VxWorks兼容 
    # bootz命令详解（ARM 32位）
# 语法: bootz <kernel_addr> <initrd_addr|-> <dtb_addr>

# 示例1: 从eMMC FAT分区启动（FAT分区）
=> fatload mmc 0:1 0x80800000 zImage        # 加载内核到0x80800000
=> fatload mmc 0:1 0x83000000 imx6q-sabresd.dtb  # 加载DTB到0x83000000
=> bootz 0x80800000 - 0x83000000            # 启动（无initrd）

# 示例2: 从eMMC EXT4分区启动
=> ext4load mmc 0:2 0x80800000 /boot/zImage
=> ext4load mmc 0:2 0x83000000 /boot/imx6q-sabresd.dtb
=> bootz 0x80800000 - 0x83000000

# 示例3: 从TFTP网络启动（开发调试）
=> dhcp 0x80800000 zImage                   # DHCP获取IP + TFTP下载
=> tftpboot 0x83000000 imx6q-sabresd.dtb
=> bootz 0x80800000 - 0x83000000

# 示例4: 使用initramfs
=> fatload mmc 0:1 0x83800000 initramfs.cpio.gz
=> bootz 0x80800000 0x83800000 0x83000000
# 注意: initrd参数是<addr>:<size>或直接地址

    本节总结¶

    核心要点回顾¶

    U-Boot关键认知：
1. U-Boot启动分两大阶段：board_init_f（Flash环境）→ relocate → board_init_r（RAM环境）
2. 环境变量存储bootcmd和bootargs，是启动配置的核心
3. bootcmd定义自动启动行为，bootargs传递给内核
4. U-Boot在跳转到内核前会修改DTB（插入bootargs和内存信息）
5. bootz/bootm/booti分别对应不同的内核格式

    关键决策清单¶

    1. 环境变量可靠性：启用CONFIG_ENV_REDUND防止CRC损坏导致启动失败
    2. bootcmd正确性：确保默认环境变量也指向正确的分区和文件
    3. DTB传递：确认U-Boot修改后的DTB中/chosen/bootargs正确
    4. 启动延迟优化：生产环境设置bootdelay=0，开发环境保留3秒
